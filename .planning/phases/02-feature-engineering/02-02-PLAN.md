  <action>

    /**
     * lib/macro-engine/features/z-scores.ts
     *
     * Point-in-time rolling z-score primitive.
     * CRITICAL: never compute z-scores across the full series — always filter to asOfDate first.
     */
    import { mean, standardDeviation } from 'simple-statistics';

    export const DAILY_WINDOW = 252;   // 1 trading year
    export const MONTHLY_WINDOW = 60;  // 5 years
    export const MIN_OBSERVATIONS = 20;

    export function rollingZScore(
      series: { date: Date; value: number }[],
      windowSize: number,
      asOfDate: Date
    ): number | null {
      const available = series
        .filter(r => r.date <= asOfDate)
        .sort((a, b) => a.date.getTime() - b.date.getTime());

      if (available.length < MIN_OBSERVATIONS) return null;

      const window = available.slice(-windowSize);
      const lookback = window.slice(0, -1);  // distribution: exclude current observation
      const current = window[window.length - 1].value;

      if (lookback.length < 5) return null;

      const values = lookback.map(r => r.value);
      const mu = mean(values);
      const sigma = standardDeviation(values);

      return sigma === 0 ? null : (current - mu) / sigma;
    }
    ```

    Also run: `npm install simple-statistics` if not already installed (check package.json first).

    // lib/macro-engine/features/factors/growth.ts
    import { subMonths } from 'date-fns';
    import { getFredAsOf, getOecdCli } from '../../query';
    import { rollingZScore, MONTHLY_WINDOW } from '../z-scores';

    export async function computeGrowthFactor(
      asOfDate: Date,
      countryCode: string | null
    ): Promise<{ value: number | null; sourceMaxDate: Date | null }> {
      if (countryCode !== null && countryCode !== 'US') {
        // Country-level: use OECD CLI as growth leading indicator
        const startDate = subMonths(asOfDate, 65);
        const rows = await getOecdCli(countryCode, startDate, asOfDate);
        if (rows.length === 0) return { value: null, sourceMaxDate: null };
        const series = rows.map(r => ({ date: r.period, value: r.cliValue }));
        const sourceMaxDate = rows[rows.length - 1].period;
        return { value: rollingZScore(series, MONTHLY_WINDOW, asOfDate), sourceMaxDate };
      }
      // US: INDPRO industrial production MoM change
      const series: { date: Date; value: number }[] = [];
      let sourceMaxDate: Date | null = null;
      for (let i = 65; i >= 0; i--) {
        const obsDate = subMonths(asOfDate, i);
        const row = await getFredAsOf('INDPRO', obsDate, asOfDate);
        if (row) {
          series.push({ date: row.observationDate, value: row.value });
          if (!sourceMaxDate || row.observationDate > sourceMaxDate) sourceMaxDate = row.observationDate;
        }
      }
      // Compute MoM % change series
      const momSeries: { date: Date; value: number }[] = [];
      for (let i = 1; i < series.length; i++) {
        const prev = series[i - 1].value;
        if (prev !== 0) {
          momSeries.push({ date: series[i].date, value: (series[i].value / prev - 1) * 100 });
        }
      }
      return { value: rollingZScore(momSeries, MONTHLY_WINDOW, asOfDate), sourceMaxDate };
    }
    ```
