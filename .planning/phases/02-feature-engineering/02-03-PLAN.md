    Strategy: Use the static approach documented in Phase 2 research. Country-health pillar scores come from World Bank annual data — for MVP, use current scores for all historical dates. Flag with countryHealthVintage = current World Bank year.

    The function should:
    1. Call the existing country-health World Bank data fetch (same path as app/api/dashboard/country-health/) to get RawVariableRow data
    2. Call scoreCountries(rawRows, populations) from lib/country-health/scoring.ts
    3. Find the CountryScore for the requested countryCode
    4. Return coreScore / 100 (normalize to 0–1) as the factor value

    Since World Bank data is fetched at runtime, memoize the result in a module-level cache (Map keyed by ISO2 + year) to avoid re-fetching for every date in a batch build.

    ```typescript
    // lib/macro-engine/features/factors/country-health.ts
    import { scoreCountries } from '../../../country-health/scoring';
    import type { RawVariableRow } from '../../../country-health/scoring';

    // Module-level cache: { countryCode -> { score: number, vintage: string } }
    const scoreCache = new Map<string, { score: number | null; vintage: string }>();

    export async function computeCountryHealthScore(
      countryCode: string
    ): Promise<{ value: number | null; sourceMaxDate: Date | null; vintage: string | null }> {
      if (scoreCache.has(countryCode)) {
        const cached = scoreCache.get(countryCode)!;
        return { value: cached.score, sourceMaxDate: null, vintage: cached.vintage };
      }

      // Fetch World Bank raw data using the same fetch pattern as the country-health route
      // Import the fetch helper from the existing country-health pipeline
      // (do not duplicate the fetch logic — import it)
      const { fetchWorldBankRows, fetchPopulations } = await import('../../../country-health/world-bank');
      const [rawRows, populations] = await Promise.all([fetchWorldBankRows(), fetchPopulations()]);
      const scores = scoreCountries(rawRows, populations);
      const currentYear = String(new Date().getFullYear() - 1); // WB data lags 1 year

      for (const cs of scores) {
        const normalized = cs.coreScore !== null ? cs.coreScore / 100 : null;
        scoreCache.set(cs.country, { score: normalized, vintage: currentYear });
      }

      const cached = scoreCache.get(countryCode);
      if (!cached) {
        scoreCache.set(countryCode, { score: null, vintage: currentYear });
        return { value: null, sourceMaxDate: null, vintage: currentYear };
      }
      return { value: cached.score, sourceMaxDate: null, vintage: cached.vintage };
    }
    ```

If no standalone fetch exists, add a minimal export to the existing fetch module — do NOT duplicate the HTTP call logic.

    For sector ETFs (countryCode = null): the caller (Plan 04) will pass null and skip calling this function. This function only handles non-null countryCode.

    ```typescript
    // lib/macro-engine/features/factors/flows-regime.ts
    import { subDays } from 'date-fns';
    import { getOhlcv } from '../../query';
    import type { OhlcvDailyRow } from '../../types';

    const VIXY_INCEPTION = new Date('2011-01-03');
    const RISK_TICKERS = ['SOXX', 'IGV', 'XLY', 'XLP', 'HYG', 'SPY'];

    function fiveDay(rows: OhlcvDailyRow[], asOfDate: Date): number | null {
      const sorted = rows.filter(r => r.date <= asOfDate).sort((a, b) => b.date.getTime() - a.date.getTime());
      if (sorted.length < 6) return null;
      const recent = sorted[0].adjClose;
      const prior  = sorted[5].adjClose;
      if (prior === 0) return null;
      return (recent / prior - 1) * 100;
    }

    function scoreSignal(ret: number | null, direction: 'up_good' | 'down_good', thresholds: [number, number, number]): number | null {
      if (ret === null) return null;
      const [lo, mid, hi] = thresholds;
      const val = direction === 'down_good' ? -ret : ret;
      if (val >= hi) return 3;
      if (val >= mid) return 2;
      if (val >= lo) return 1;
      return 0;
    }

    export async function computeFlowsRegimeScore(
      asOfDate: Date
    ): Promise<{ value: number | null; sourceMaxDate: Date | null }> {
      const windowStart = subDays(asOfDate, 30);
      const vixTicker = asOfDate >= VIXY_INCEPTION ? 'VIXY' : 'VXX';

      const allTickers = [vixTicker, ...RISK_TICKERS];
      const seriesMap: Record<string, OhlcvDailyRow[]> = {};
      let sourceMaxDate: Date | null = null;

      for (const t of allTickers) {
        const rows = await getOhlcv(t, windowStart, asOfDate);
        seriesMap[t] = rows;
        for (const r of rows) {
          if (!sourceMaxDate || r.date > sourceMaxDate) sourceMaxDate = r.date;
        }
      }

      // Score each signal (replicates buildRegime() logic from flows/route.ts)
      const vixRet   = fiveDay(seriesMap[vixTicker], asOfDate);
      const semiRet  = fiveDay(seriesMap['SOXX'], asOfDate);
      const softRet  = fiveDay(seriesMap['IGV'], asOfDate);
      const cyclRet  = fiveDay(seriesMap['XLY'], asOfDate);
      const defRet   = fiveDay(seriesMap['XLP'], asOfDate);
      const hygRet   = fiveDay(seriesMap['HYG'], asOfDate);
      const spyRet   = fiveDay(seriesMap['SPY'], asOfDate);

      const semiSpread = semiRet !== null && softRet !== null ? semiRet - softRet : null;
      const cyclSpread = cyclRet !== null && defRet !== null  ? cyclRet - defRet  : null;

      const scores = [
        scoreSignal(vixRet,    'down_good', [-2, -5, -10]),  // VIXY: falling VIX = risk-on
        scoreSignal(semiSpread,'up_good',   [0,  1,  2]),
        scoreSignal(cyclSpread,'up_good',   [0,  0.5, 1.5]),
        scoreSignal(hygRet,    'up_good',   [0,  0.3, 0.8]),
        scoreSignal(spyRet,    'up_good',   [0,  0.5, 1.5]),
      ].filter(s => s !== null) as number[];

      if (scores.length < 3) return { value: null, sourceMaxDate };  // insufficient signals

      const total = scores.reduce((a, b) => a + b, 0);
      const maxPossible = scores.length * 3;
      return { value: total / maxPossible, sourceMaxDate };
    }
    ```
